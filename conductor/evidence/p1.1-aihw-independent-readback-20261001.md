```python
from bs4 import BeautifulSoup

def extract(path):
    soup = BeautifulSoup(open(path, "rb").read(), "html.parser")
    table = next(t for t in soup.find_all("table")
                 if t.caption and t.caption.get_text(" ", strip=True)
                 == "Table 2: Emergency department presentations by triage category, 2024–25")
    rows = [[c.get_text(" ", strip=True) for c in tr.find_all(["th", "td"])]
            for tr in table.find_all("tr")]
    presentations = rows[1]
    categories = [int(value.replace(",", "")) for value in presentations[1:-1]]
    total = int(presentations[-1].replace(",", ""))
    return rows, total, sum(categories), total - sum(categories)

for path in (".artifacts/sources/aihw-hospitals-at-a-glance-independent.html",
             ".artifacts/sources/aihw-hospitals-at-a-glance-2024-25.html"):
    rows, total, subtotal, difference = extract(path)
    print(path, rows, total, subtotal, difference)
```
